        function selectDifficulty(element, value) {
            // Remove active class from all difficulty cards
            const cards = element.parentElement.querySelectorAll('.step-card');
            cards.forEach(card => {
                card.classList.remove('active');
                const icon = card.querySelector('.material-symbols-outlined');
                const text = card.querySelector('h3');
                icon.classList.replace('text-primary', 'text-tertiary');
                text.classList.remove('text-primary');
            });

            // Add active class to selected
            element.classList.add('active');
            const activeIcon = element.querySelector('.material-symbols-outlined');
            const activeText = element.querySelector('h3');
            activeIcon.classList.replace('text-tertiary', 'text-primary');
            activeText.classList.add('text-primary');

            // Update summary
            document.getElementById('summary-difficulty').textContent = value;
            document.getElementById('summary-difficulty').classList.add('text-primary');
        }

        function switchTab(tabId) {

                el.classList.remove('text-primary', 'border-b-2', 'border-primary');
                el.classList.add('text-on-surface-variant');
            });

            document.getElementById('section-' + tabId).classList.remove('hidden');

            const activeTab = document.getElementById('tab-' + tabId);
